Python Starter Projects

Pick a starter, get it running, then change it to make it yours. Every file has comments explaining the code and level-up ideas at the bottom.

How to use a starter

  1. Pick a project below and click Copy code.
  2. Paste it into your Python editor and press Run.
  3. Read the comments (lines that start with #) to see what each part does.
  4. Change one thing at a time: a color, a speed, a question. Run it again to see what happened.
  5. Try the Level up ideas at the bottom of the file.

Where to run: the Quiz app works in any Python editor. Pong, Snake, and Paint use turtle graphics with the keyboard and mouse, so run them in IDLE (comes with Python on a laptop) or another editor that supports turtle.

# QUIZ APP STARTER
# Ask questions, check answers, and keep score.
# Runs in any Python editor, no extra setup needed.

# Each question is a pair: (question, answer)
# TRY THIS: change these to your own questions!
questions = [
    ("What is 7 x 8? ", "56"),
    ("What planet is closest to the sun? ", "mercury"),
    ("How many legs does a spider have? ", "8"),
]

score = 0

print("Welcome to my quiz!")
name = input("What is your name? ")

for question, answer in questions:
    guess = input(question)
    # .lower() and .strip() so "Mercury " still counts as "mercury"
    if guess.lower().strip() == answer:
        print("Correct!")
        score = score + 1
    else:
        print("Not quite. The answer was", answer)

print(name + ", you got", score, "out of", len(questions))

Pong

Bounce the ball with your paddle. Up and Down arrow keys. Uses turtle graphics.

Download 
# PONG STARTER
# Keep the ball from getting past your paddle.
# Controls: Up and Down arrow keys.
# Uses turtle, which comes with Python.

import turtle

# --- Set up the screen ---
screen = turtle.Screen()
screen.title("My Pong")
screen.bgcolor("black")
screen.setup(width=800, height=600)
screen.tracer(0)  # we redraw the screen ourselves, so it runs smoothly

# --- The paddle ---
paddle = turtle.Turtle()
paddle.shape("square")
paddle.color("white")
paddle.shapesize(stretch_wid=5, stretch_len=1)  # tall and thin
paddle.penup()
paddle.goto(-350, 0)

# --- The ball ---
ball = turtle.Turtle()
ball.shape("circle")
ball.color("yellow")
ball.penup()
ball.dx = 3   # how far the ball moves sideways each step
ball.dy = 3   # how far the ball moves up/down each step

# --- The score ---
score = 0
pen = turtle.Turtle()
pen.color("white")
pen.penup()
pen.hideturtle()
pen.goto(0, 260)

def show_score():
    pen.clear()
    pen.write("Score: " + str(score), align="center", font=("Courier", 24, "bold"))

show_score()

# --- Moving the paddle ---
def paddle_up():
    if paddle.ycor() < 250:
        paddle.sety(paddle.ycor() + 30)

def paddle_down():
    if paddle.ycor() > -250:
        paddle.sety(paddle.ycor() - 30)

screen.listen()
screen.onkeypress(paddle_up, "Up")
screen.onkeypress(paddle_down, "Down")

# --- The game loop: runs over and over ---
def game_loop():
    global score
    ball.setx(ball.xcor() + ball.dx)
    ball.sety(ball.ycor() + ball.dy)

    # Bounce off the top and bottom
    if ball.ycor() > 290 or ball.ycor() < -290:
        ball.dy = -ball.dy

    # Bounce off the right wall
    if ball.xcor() > 390:
        ball.dx = -ball.dx

    # Hit the paddle!
    if -350 < ball.xcor() < -330 and abs(ball.ycor() - paddle.ycor()) < 55:
        ball.dx = -ball.dx
        score = score + 1
        show_score()

    # Missed! Start the ball again from the middle
    if ball.xcor() < -390:
        ball.goto(0, 0)
        ball.dx = 3
        score = 0
        show_score()

    screen.update()
    screen.ontimer(game_loop, 16)  # about 60 times a second

game_loop()
turtle.done()

# LEVEL UP IDEAS
# 1. Make the ball faster each time you hit it (try ball.dx = ball.dx * 1.1).
# 2. Change the colors or make the paddle smaller for a harder game.
# 3. Add a second paddle on the right for a friend (use the W and S keys).
# 4. Show "Game Over" after 3 misses.

Snake

Eat food, grow longer, don't crash. Arrow keys. Uses turtle graphics.

Download 
# SNAKE STARTER
# Eat the food to grow longer. Don't hit the walls or yourself!
# Controls: arrow keys.
# Uses turtle, which comes with Python.

import turtle
import random

STEP = 20  # size of one square

# --- Set up the screen ---
screen = turtle.Screen()
screen.title("My Snake")
screen.bgcolor("darkgreen")
screen.setup(width=600, height=600)
screen.tracer(0)

# --- The snake's head ---
head = turtle.Turtle()
head.shape("square")
head.color("lightgreen")
head.penup()
head.direction = "stop"

body = []  # the rest of the snake goes in this list

# --- The food ---
food = turtle.Turtle()
food.shape("circle")
food.color("red")
food.penup()

def move_food():
    food.goto(random.randint(-14, 14) * STEP, random.randint(-14, 14) * STEP)

move_food()

# --- The score ---
score = 0
pen = turtle.Turtle()
pen.color("white")
pen.penup()
pen.hideturtle()
pen.goto(0, 260)

def show_score():
    pen.clear()
    pen.write("Score: " + str(score), align="center", font=("Courier", 20, "bold"))

show_score()

# --- Steering (no turning straight back into yourself) ---
def go_up():
    if head.direction != "down":
        head.direction = "up"

def go_down():
    if head.direction != "up":
        head.direction = "down"

def go_left():
    if head.direction != "right":
        head.direction = "left"

def go_right():
    if head.direction != "left":
        head.direction = "right"

screen.listen()
screen.onkeypress(go_up, "Up")
screen.onkeypress(go_down, "Down")
screen.onkeypress(go_left, "Left")
screen.onkeypress(go_right, "Right")

def move_head():
    if head.direction == "up":
        head.sety(head.ycor() + STEP)
    if head.direction == "down":
        head.sety(head.ycor() - STEP)
    if head.direction == "left":
        head.setx(head.xcor() - STEP)
    if head.direction == "right":
        head.setx(head.xcor() + STEP)

def reset_game():
    global score
    head.goto(0, 0)
    head.direction = "stop"
    for part in body:
        part.hideturtle()
    body.clear()
    score = 0
    show_score()

# --- The game loop ---
def game_loop():
    global score

    # Eat the food: grow one square longer
    if head.distance(food) < STEP:
        move_food()
        part = turtle.Turtle()
        part.shape("square")
        part.color("green")
        part.penup()
        body.append(part)
        score = score + 1
        show_score()

    # Each body part moves to where the part in front of it was
    for i in range(len(body) - 1, 0, -1):
        body[i].goto(body[i - 1].xcor(), body[i - 1].ycor())
    if body:
        body[0].goto(head.xcor(), head.ycor())

    move_head()

    # Hit a wall?
    if abs(head.xcor()) > 290 or abs(head.ycor()) > 290:
        reset_game()

    # Hit yourself?
    for part in body:
        if part.distance(head) < STEP / 2:
            reset_game()
            break

    screen.update()
    screen.ontimer(game_loop, 120)  # smaller number = faster snake

game_loop()
turtle.done()

# LEVEL UP IDEAS
# 1. Make the snake speed up as it gets longer.
# 2. Keep a high score that stays when you restart.
# 3. Add golden food worth 3 points.
# 4. Let the snake wrap around to the other side instead of hitting walls.

Paint app

Drag the mouse to draw. Keys change colors and brush size. Uses turtle graphics.

# PAINT APP STARTER
# Drag with your mouse to draw. Press keys to change colors.
# Keys: r = red, g = green, b = blue, k = black
#       up/down arrows = bigger/smaller brush, c = clear
# Uses turtle, which comes with Python.

import turtle

screen = turtle.Screen()
screen.title("My Paint App")
screen.bgcolor("white")
screen.setup(width=800, height=600)

brush = turtle.Turtle()
brush.shape("circle")
brush.speed(0)       # draw as fast as possible
brush.pensize(5)
brush.color("black")

# --- Drawing with the mouse ---
def draw(x, y):
    brush.ondrag(None)   # pause while we move, so it doesn't lag
    brush.goto(x, y)
    brush.ondrag(draw)

def jump(x, y):
    # Clicking somewhere moves the brush there without drawing a line
    brush.penup()
    brush.goto(x, y)
    brush.pendown()

brush.ondrag(draw)
screen.onclick(jump)

# --- Changing colors ---
def red():
    brush.color("red")

def green():
    brush.color("green")

def blue():
    brush.color("blue")

def black():
    brush.color("black")

# --- Brush size ---
def bigger():
    brush.pensize(brush.pensize() + 2)

def smaller():
    if brush.pensize() > 2:
        brush.pensize(brush.pensize() - 2)

def clear():
    brush.clear()

screen.listen()
screen.onkey(red, "r")
screen.onkey(green, "g")
screen.onkey(blue, "b")
screen.onkey(black, "k")
screen.onkey(bigger, "Up")
screen.onkey(smaller, "Down")
screen.onkey(clear, "c")

turtle.done()

# LEVEL UP IDEAS
# 1. Add more colors (try "purple", "orange", or "pink").
# 2. Add an eraser: a key that sets the color to "white".
# 3. Press s to stamp a shape where the brush is (look up brush.stamp()).
# 4. Change the background color with a key.
